use std::env;
use std::fs::OpenOptions;
use std::io::Write;

fn mark_probe() {
    let Some(path) = env::var_os("BITCODE_ORACLE_PROBE") else {
        return;
    };
    let mut probe = OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)
        .expect("open oracle probe");
    writeln!(probe, "selected_operation").expect("write oracle probe");
}

pub fn selected_operation() -> &'static str {
    mark_probe();
    "changed"
}

pub fn unrelated_operation() -> &'static str {
    "unrelated"
}

/// RAII guard whose Drop reaches `selected_operation` only through the
/// implicit destructor call, not an ordinary function call.
pub struct Guard;

impl Guard {
    pub fn new() -> Self {
        Guard
    }
}

impl Drop for Guard {
    fn drop(&mut self) {
        let _ = selected_operation();
    }
}
